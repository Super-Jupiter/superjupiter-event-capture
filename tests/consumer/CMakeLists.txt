cmake_minimum_required(VERSION 3.16)
project(SuperJupiterEventCaptureConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_EVENT_CAPTURE_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_EVENT_CAPTURE_SOURCE_DIR}" event-capture)
  if(TARGET superjupiter-event-capture-tests OR TARGET superjupiter-event-capture-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterEventCapture 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-event-capture-consumer main.cpp)
target_link_libraries(superjupiter-event-capture-consumer PRIVATE SuperJupiter::EventCapture)
